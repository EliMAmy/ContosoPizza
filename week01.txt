. A text copy of your working sales summary function from the Work with files and directories in a .NET app module.

void GenerateSalesSummary(
    IEnumerable<string> salesFiles,
    double grandTotal)
{
    var report = new List<string>();

    report.Add("Sales Summary");
    report.Add("-----------------");
    report.Add($"Total Sales: {grandTotal:C}");
    report.Add("");
    report.Add("Details:");

    foreach (var file in salesFiles)
    {
        string salesJson = File.ReadAllText(file);

        SalesData? data =
            JsonConvert.DeserializeObject<SalesData>(salesJson);

        double fileTotal = data?.Total ?? 0;

        string fileName = Path.GetRelativePath(storesDirectory, file);

        report.Add($"{fileName}: {fileTotal:C}");
    }

    string reportPath = Path.Combine(
        salesTotalDir,
        "SalesSummary.txt"
    );

    File.WriteAllLines(reportPath, report);

    Console.WriteLine("Sales summary created successfully!");
}
